Order Summary

JavaScript Solution


          const orderJson = `{
            "customer": "Sam",
            "items": [
              { "name": "Notebook", "price": 4.50, "quantity": 2 },
              { "name": "Pen", "price": 1.25, "quantity": 3 },
              { "name": "Backpack", "price": 28.00, "quantity": 1 }
            ]
          }`;

          const data = JSON.parse(orderJson);
          let completeOrderTotal = 0;
          const expensiveItems = [];

          for(let i = 0; i < data.items.length; i++) {
            const item = data.items[i];
            const itemTotal = item.price * item.quantity;

            completeOrderTotal += itemTotal;

            if(itemTotal > 5) {
              expensiveItems.push({
                name: item.name,
                total: itemTotal
              });
            }
          }

          /* Output to HTML */
          const summary = document.getElementById("order-summary");
          let output = `<div class="summary-item"><span>Customer</span><strong>${data.customer}</strong></div>`;
          output += `<div class="summary-item"><span>Order Total</span><strong>£${completeOrderTotal.toFixed(2)}</strong></div>`;
          output += '<div class="summary-item summary-item-wide"><span>Items over £5</span><ul>';

          for (const item of expensiveItems) {
            output += `<li>${item.name}: £${item.total.toFixed(2)}</li>`;
          }

          output += "</ul></div>";

          summary.innerHTML = output;
