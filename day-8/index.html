Daily Challenges: Day 8 - JavaScript + JSON

Estimated time: 30 minutes

Level: Beginner to intermediate

Objective: Practise parsing JSON and using Array.reduce() to validate records and build a summary.

Valid Order Summary

Valid Orders 0
Skipped Orders 0
Total Items 0
Total Revenue £0.00

JavaScript Solution

    const orderJson = `[
      {"customer":"Maya","quantity":2,"price":12.50},
      {"customer":"Leo","quantity":1,"price":20},
      {"customer":"Maya","quantity":3,"price":5},
      {"customer":"","quantity":2,"price":8},
      {"customer":"Nora","quantity":0,"price":15}
    ]`;
      
    function summariseOrders(ordersJson) {
      const data = JSON.parse(ordersJson);
    
      const summary = data.reduce((totals, order) => {
        const isValid =
          order.customer !== "" &&
          order.quantity > 0 &&
          order.price > 0;
    
        if (isValid) {
          totals.validOrders++;
          totals.totalItems = totals.totalItems + order.quantity;
          totals.totalRevenue = totals.totalRevenue + (order.quantity * order.price);
        } else {
          totals.skippedOrders++;
        }
    
        return totals;
      }, {
        validOrders: 0,
        skippedOrders: 0,
        totalItems: 0,
        totalRevenue: 0
      });
    
      return summary;
    }
      
    const result = summariseOrders(orderJson);
    console.log(result);
