          function calculateBasket(items) {
            let subtotal = 0;
            let discount = 0;

            const itemsTable = document.querySelector("#items");

            for (const item of items) {
              const itemTotal = item.price * item.quantity;
              const row = itemsTable.insertRow();

              row.insertCell().textContent = item.name;
              row.insertCell().textContent = `£${item.price.toFixed(2)}`;
              row.insertCell().textContent = item.quantity;
              row.insertCell().textContent = `£${itemTotal.toFixed(2)}`;

              subtotal += itemTotal;
            }

            if(subtotal >= 20) {
              discount = subtotal * 0.10;
            } else {
              discount = 0;
            }

            const total = subtotal - discount;

            return {
              subtotal: Number(subtotal.toFixed(2)),
              discount: Number(discount.toFixed(2)),
              total: Number(total.toFixed(2))
            };
          }

          const items = [
            { name: "Notebook", price: 4.5, quantity: 2 },
            { name: "Pen", price: 1.25, quantity: 3 },
            { name: "Backpack", price: 18, quantity: 1 }
          ];

          const result = calculateBasket(items);

          /* Output to HTML */
          document.querySelector("#subtotal").textContent = result.subtotal.toFixed(2);

          document.querySelector("#discount").textContent = result.discount.toFixed(2);

          document.querySelector("#total").textContent = result.total.toFixed(2);
